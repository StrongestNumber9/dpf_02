129 public void collect(Dataset<Row> batchDF, Long batchId, List<AbstractStep> postBcSteps, boolean skipLimiting) {
130 // Apply post-batchcollect limit if steps are present, otherwise use the default.
131 // limit<=0 means no limit
132 final int limit;
133 if (!postBcSteps.isEmpty()) {
134 limit = this.postBcLimit;
135 } else {
136 limit = this.defaultLimit;
137 }
138
139 // check that sortColumn (_time) exists,
140 // and get the sortColId
141 // otherwise, no sorting will be done.
142 if (this.inputSchema == null) {
143 this.inputSchema = batchDF.schema();
144 }
145
146 if (this.getListOfSortByClauses() == null || this.getListOfSortByClauses().isEmpty()) {
147 for (String field : this.inputSchema.fieldNames()) {
148 if (field.equals(this.sortColumn)) {
149 this.sortedBySingleColumn = true;
150 break;
151 }
152 }
153 }
154
155 Dataset<Row> orderedDs = orderDataset(batchDF);
156 if (!skipLimiting && limit > 0) {
157 orderedDs = orderedDs.limit(limit);
158 }
159 List<Row> collected = orderedDs.collectAsList();
160 Dataset<Row> createdDsFromCollected = SparkSession.builder().getOrCreate().createDataFrame(collected, this.inputSchema);
161 Dataset<Row> current;
162 if (this.savedDs == null) {
163 current = createdDsFromCollected;
164 }
165 else {
166 current = savedDs.union(createdDsFromCollected);
167 }
168
169 current = orderDataset(current);
170 if (!skipLimiting && limit > 0) {
171 current = current.limit(limit);
172 }
173 this.savedDs = current;
174
175 // Post batchCollect steps processing
176 Dataset<Row> rv = current;
177 for (final AbstractStep step : postBcSteps) {
178 try {
179 rv = step.get(rv);
180 } catch (StreamingQueryException e) {
181 throw new IllegalStateException("Exception occurred while running post-batchcollect steps: ", e);
182 }
183 }
184
185 this.outputDs = rv;
186 }
