80 // Performs orderBy operation on a dataset and returns the ordered one
81 public Dataset<Row> orderDatasetByGivenColumns(Dataset<Row> ds) {
82 if (!this.listOfSortByClauses.isEmpty()) {
83 ArrayList<Column> sortColumns = new ArrayList<>();
84 for (SortByClause sbc : listOfSortByClauses) {
85 if (sbc.getSortAsType() == SortByClause.Type.AUTOMATIC) {
86 sbc = detectSortByType(ds, sbc);
87 }
88 sortColumns.add(sortByClauseToColumn(sbc));
89 }
90 ds = ds.orderBy(JavaConversions.asScalaBuffer(sortColumns));
91 }
92
93 return ds;
94 }

96 // make Column by sortByClause type and if it is descending/ascending
97 private Column sortByClauseToColumn(SortByClause sortByClause) {
98 Column column;
99 switch (sortByClause.getSortAsType()) {
100 case STRING:
101 column = sortByClause.isDescending() ?
102 functions.col(sortByClause.getFieldName()).cast(DataTypes.StringType).desc() :
103 functions.col(sortByClause.getFieldName()).cast(DataTypes.StringType).asc();
104 break;
105 case NUMERIC:
106 column = sortByClause.isDescending() ?
107 functions.col(sortByClause.getFieldName()).cast(DataTypes.DoubleType).desc() :
108 functions.col(sortByClause.getFieldName()).cast(DataTypes.DoubleType).asc();
109 break;
110 case IP_ADDRESS:
111 final SparkSession ss = SparkSession.builder().getOrCreate();
112 UserDefinedFunction ipStringToIntUDF = functions.udf(new ConvertIPStringToInt(), DataTypes.LongType);
113 ss.udf().register("ip_string_to_int", ipStringToIntUDF);
114 Column sortingCol = functions.callUDF("ip_string_to_int", functions.col(sortByClause.getFieldName()));
115
116 column = sortByClause.isDescending() ? sortingCol.desc() : sortingCol.asc();
117 break;
118 case DEFAULT:
119 default:
120 column = sortByClause.isDescending() ?
121 functions.col(sortByClause.getFieldName()).desc() :
122 functions.col(sortByClause.getFieldName()).asc();
123 }
124 return column;
125 }

127 // detect sorting type if auto() was used in sort
128 private SortByClause detectSortByType(final Dataset<Row> ds, final SortByClause sbc) {
129 StructField[] fields = ds.schema().fields();
130 SortByClause newSbc = new SortByClause();
131 newSbc.setDescending(sbc.isDescending());
132 newSbc.setLimit(sbc.getLimit());
133 newSbc.setFieldName(sbc.getFieldName());
134
135 for (StructField field : fields) {
136 if (field.name().equals(sbc.getFieldName())) {
137 switch (field.dataType().typeName()) {
138 case "string": // ip address?
139 newSbc.setSortAsType(numericalStringCheck(ds, sbc.getFieldName()));
140 break;
141 case "long":
142 case "integer":
143 case "float":
144 case "double":
145 newSbc.setSortAsType(SortByClause.Type.NUMERIC);
146 break;
147 case "timestamp":
148 newSbc.setSortAsType(SortByClause.Type.NUMERIC); // convert to unix epoch?
149 break;
150 default:
151 newSbc.setSortAsType(SortByClause.Type.DEFAULT);
152 }
153 }
154 }
155 return newSbc;
156 }

164 private SortByClause.Type numericalStringCheck(Dataset<Row> dataset, String fieldName) {
165 // Value is numerical if it is castable to Double
166 Dataset<Row> tempDataset = dataset.withColumn("isNumerical", functions.col(fieldName).cast(DataTypes.DoubleType).isNotNull());
167
168 // If the isNumerical column has even one false value, the column contains strings
169 if (tempDataset.filter(tempDataset.col("isNumerical").contains(false)).isEmpty()) {
170 // no false values found
171 return SortByClause.Type.NUMERIC;
172 } else {
173 return SortByClause.Type.STRING;
174 }
175 }
